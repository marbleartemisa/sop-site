const SHEET_ID = "1zUZ37cWCS4N72AG5H2c_RRgjGhcQ5pj_D9qNBbRCRcI"; const SHEET_NAME = "SOP_Admin"; const SHEET_APORTES = "SOP_Aportes"; const ESTADO_COL = 7; /* ========================= GET DATOS ========================= */ function doGet(e) { const ss = SpreadsheetApp.openById(SHEET_ID); /* HISTORIAL DE APORTES */ if (e.parameter.aportes) { const sheetAportes = ss.getSheetByName(SHEET_APORTES); if (!sheetAportes) return ContentService.createTextOutput("[]").setMimeType(ContentService.MimeType.JSON); const data = sheetAportes.getDataRange().getValues(); if (data.length < 2) return ContentService.createTextOutput("[]").setMimeType(ContentService.MimeType.JSON); data.shift(); // eliminar cabecera const id = e.parameter.aportes; const aportes = data .filter(r => r[1] == id) // r[1] = id_sop .map(r => ({ EMPLEADO: r[2], COMENTARIO: r[5], ARCHIVOS: r[6], FECHA: r[7] })); return ContentService.createTextOutput(JSON.stringify(aportes)).setMimeType(ContentService.MimeType.JSON); } /* LISTA DE SOP */ const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) return ContentService.createTextOutput(JSON.stringify({ success: false, error: "No se encontró la hoja" })).setMimeType(ContentService.MimeType.JSON); const data = sheet.getDataRange().getValues(); if (data.length < 2) return ContentService.createTextOutput("[]").setMimeType(ContentService.MimeType.JSON); data.shift(); // eliminar cabecera const registros = data.map((row, index) => ({ id: row[8] || (new Date().getTime() + index), // Timestamp único empleado: row[0], correo: row[1], departamento: row[2], objetivo: row[3], areas: row[4], resumen: row[5], estado: row[6] || "Pendiente", links: row[7] || "", Timestamp: row[8] || new Date().toISOString() })); return ContentService.createTextOutput(JSON.stringify(registros)).setMimeType(ContentService.MimeType.JSON); } /* ========================= POST ========================= */ function doPost(e) { const data = JSON.parse(e.postData.contents); /* GUARDAR APORTE */ if (data.action === "guardarAporte") { const resultado = guardarAporte(data); return ContentService.createTextOutput(JSON.stringify(resultado)).setMimeType(ContentService.MimeType.JSON); } const ss = SpreadsheetApp.openById(SHEET_ID); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) return ContentService.createTextOutput(JSON.stringify({ success: false, error: "No se encontró la hoja" })).setMimeType(ContentService.MimeType.JSON); const fila = Number(data.id); if (isNaN(fila) || fila < 2) return ContentService.createTextOutput(JSON.stringify({ success: false, error: "Fila inválida" })).setMimeType(ContentService.MimeType.JSON); /* BORRAR */ if (data.action === "delete") { sheet.deleteRow(fila); return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON); } /* CAMBIAR ESTADO */ if (data.estado !== undefined) { const nuevoEstado = String(data.estado); sheet.getRange(fila, ESTADO_COL).setValue(nuevoEstado); return ContentService.createTextOutput(JSON.stringify({ success: true })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Parámetros inválidos" })).setMimeType(ContentService.MimeType.JSON); } /* ========================= GUARDAR APORTE ========================= */ function guardarAporte(data) { const ss = SpreadsheetApp.openById(SHEET_ID); const sheetSOP = ss.getSheetByName(SHEET_NAME); const sheetAportes = ss.getSheetByName(SHEET_APORTES); if (!sheetSOP) return { success: false, error: "No existe hoja SOP_Admin" }; if (!sheetAportes) return { success: false, error: "No existe hoja SOP_Aportes" }; // 🔹 Validar existencia del SOP por Timestamp const sopValues = sheetSOP.getDataRange().getValues(); const existe = sopValues.some(row => String(row[8]) === String(data.id_sop)); if (!existe) { return { success: false, error: "No se encontró la solicitud (ID_SOP inválido)" }; } // 🔹 Generar ID único const idAporte = "APT-" + new Date().getTime(); // 🔹 Fecha const fecha = new Date(); // 🔹 Guardar sheetAportes.appendRow([ idAporte, // ID_APORTE data.id_sop, // ID_SOP (Timestamp) data.empleado, data.correo, data.departamento, data.comentario, data.archivos || "", // links fecha ]); return { success: true, id_aporte: idAporte }; }
